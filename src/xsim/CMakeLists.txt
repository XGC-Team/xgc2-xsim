cmake_minimum_required(VERSION 3.16)
project(xsim LANGUAGES C CXX)

include(GNUInstallDirs)
set(XGC2_MATH_INCLUDE "/usr/include" CACHE PATH "Installed xgc2-math headers or explicit source include")
set(EIGEN_INCLUDE "/usr/include/eigen3" CACHE PATH "Eigen headers")
set(FS150_ASSET_SOURCE_ROOT "" CACHE PATH "Owning FS150 SITL asset source")
if(NOT EXISTS "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py")
  message(FATAL_ERROR "Set FS150_ASSET_SOURCE_ROOT; no copied physical defaults")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(asset_header "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.hpp")
set(asset_json "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py"
  --sdf "${FS150_ASSET_SOURCE_ROOT}/models/fs150/iris.sdf"
  --params "${FS150_ASSET_SOURCE_ROOT}/config/generated/fs150-sitl.params"
  --fcu-name-catalog "${CMAKE_CURRENT_SOURCE_DIR}/models/fcu_parameter_names.json"
  --output "${asset_json}.tmp" --cpp-output "${asset_header}.tmp" RESULT_VARIABLE asset_status)
if(NOT asset_status EQUAL 0)
  message(FATAL_ERROR "FS150 asset extraction failed")
endif()
configure_file("${asset_header}.tmp" "${asset_header}" COPYONLY)
configure_file("${asset_json}.tmp" "${asset_json}" COPYONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
option(XSIM_TESTS "Build the model and standalone server checks" OFF)

add_subdirectory(models/px4_v1_12_3)

find_package(Threads REQUIRED)
find_package(XgcXrpc CONFIG REQUIRED COMPONENTS http bootstrap)
find_package(XgcChassisHold CONFIG REQUIRED COMPONENTS core)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(XgcWorldLidar CONFIG REQUIRED)
option(XSIM_ROS "Build the original ROS1 IO boundary" ON)
add_library(xsim_world STATIC core/world.cpp systems/robots.cpp)
target_compile_features(xsim_world PUBLIC cxx_std_17)
target_include_directories(xsim_world PUBLIC "${XGC2_MATH_INCLUDE}" "${EIGEN_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/models")
target_link_libraries(xsim_world PUBLIC px4_v1_12_3::core XgcChassisHold::core Threads::Threads)
add_library(xsim_config STATIC io/config.cpp io/experiment.cpp)
target_link_libraries(xsim_config PUBLIC xsim_world nlohmann_json::nlohmann_json yaml-cpp)
add_library(xsim_startup STATIC io/startup.cpp)
target_compile_features(xsim_startup PUBLIC cxx_std_17)
target_include_directories(xsim_startup PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(xsim_sensors STATIC systems/sensors.cpp)
target_link_libraries(xsim_sensors PUBLIC xsim_world nlohmann_json::nlohmann_json XgcWorldLidar::Geometry)
option(XSIM_GPU "Compile the explicit original GPU backend" OFF)
if(XSIM_GPU)
  target_compile_definitions(xsim_sensors PUBLIC XSIM_GPU=1)
  target_link_libraries(xsim_sensors PUBLIC XgcWorldLidar::Gpu)
endif()
add_library(xsim_native STATIC io/native_rpc/server.cpp io/native_rpc/simulation.cpp)
target_link_libraries(xsim_native PUBLIC xsim_sensors xsim_config xsim_startup XgcXrpc::http)
add_executable(xsim main.cpp)
target_link_libraries(xsim PRIVATE xsim_native XgcXrpc::bootstrap)
# Owning sensor headers must precede older copies in the ROS installation.
# Imported includes otherwise follow explicit catkin SYSTEM paths and can give
# main/ROS a different Sensors layout from the sensor implementation.
set_property(TARGET xsim PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
if(XSIM_ROS)
  find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs nav_msgs rosgraph_msgs mavros_msgs)
  add_library(xsim_ros STATIC io/ros/entity.cpp io/ros/runtime.cpp)
  target_include_directories(xsim_ros SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(xsim_ros PUBLIC xsim_sensors ${catkin_LIBRARIES})
  target_include_directories(xsim SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
  target_compile_definitions(xsim PRIVATE XSIM_ROS=1)
  target_link_libraries(xsim PRIVATE xsim_ros)
  target_compile_options(xsim_ros PRIVATE -O2 -Wall -Wextra)
  set_property(TARGET xsim_ros PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
endif()
foreach(target xsim_world xsim_config xsim_startup xsim_sensors xsim_native xsim)
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra)
endforeach()

  foreach(test_name rigid_body flight_controller model px4_setpoint ground_contact)
    add_executable(${test_name}_test ${test_name}_test.cpp)
    target_compile_options(${test_name}_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(${test_name}_test PRIVATE xsim_world)
    add_test(NAME xsim_${test_name} COMMAND ${test_name}_test)

install(TARGETS xsim RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${asset_json}" DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim)
install(DIRECTORY ../../config/ DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/config)
install(FILES ../../README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim)
install(DIRECTORY ../../docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/docs)
install(FILES models/px4_v1_12_3/upstream/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/licenses RENAME PX4-LICENSE)
set(CPACK_PACKAGE_NAME "xsim")
set(CPACK_PACKAGE_VERSION "1.1.0")
set(CPACK_GENERATOR "TGZ")
include(CPack)
