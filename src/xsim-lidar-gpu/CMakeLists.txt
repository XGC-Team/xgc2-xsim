cmake_minimum_required(VERSION 3.16)
project(xsim-lidar-gpu LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters features search kdtree)
find_package(OpenCV REQUIRED)
find_package(OpenGL REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(XGC2_MATH_INCLUDE "/usr/include" CACHE PATH "Installed xgc2-math headers (0.6.0 or later)")
if(NOT EXISTS "${XGC2_MATH_INCLUDE}/xgc2_math/scene.hpp")
  message(FATAL_ERROR "xgc2-math 0.6.0 or later scene headers required")
endif()
find_path(XSIM_LIDAR_GPU_GLM_INCLUDE glm/glm.hpp)
if(NOT XSIM_LIDAR_GPU_GLM_INCLUDE)
  message(FATAL_ERROR "GLM headers required for the original GPU implementation")
endif()
set(upstream "${CMAKE_CURRENT_SOURCE_DIR}/upstream/local_sensing/include")
set(renderer_sources ${upstream}/glad.c ${upstream}/FOV_Checker/FOV_Checker.cpp)
set(renderer_includes "${CMAKE_CURRENT_SOURCE_DIR}/../xsim/sensors"
  "${CMAKE_CURRENT_SOURCE_DIR}/local_sensing/include" "${upstream}"
  "${XGC2_MATH_INCLUDE}" "${XSIM_LIDAR_GPU_GLM_INCLUDE}" ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
set(renderer_libraries Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS}
  ${OPENGL_LIBRARIES} glfw OpenMP::OpenMP_CXX ${CMAKE_DL_LIBS})
add_library(xsim-lidar-gpu SHARED module.cpp ${renderer_sources})
target_compile_features(xsim-lidar-gpu PRIVATE cxx_std_17)
target_include_directories(xsim-lidar-gpu PRIVATE ${renderer_includes})
target_compile_definitions(xsim-lidar-gpu PRIVATE ROOT_DIR="${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/marsim/")
target_link_libraries(xsim-lidar-gpu PRIVATE ${renderer_libraries})
set_target_properties(xsim-lidar-gpu PROPERTIES CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(BUILD_TESTING)
  add_executable(xsim_lidar_gpu_abi_test tests/abi_test.cpp)
  target_compile_features(xsim_lidar_gpu_abi_test PRIVATE cxx_std_17)
  target_compile_options(xsim_lidar_gpu_abi_test PRIVATE -UNDEBUG)
  target_include_directories(xsim_lidar_gpu_abi_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../xsim/sensors" "${XGC2_MATH_INCLUDE}")
  target_link_libraries(xsim_lidar_gpu_abi_test PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME xsim_lidar_gpu_abi COMMAND xsim_lidar_gpu_abi_test $<TARGET_FILE:xsim-lidar-gpu>)
  add_test(NAME xsim_lidar_gpu_sources COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../../.xgc2/scripts/check_gpu_module_sources.py")
  # Compiled with the suite, but execution requires real hardware and a display.
  add_executable(xsim_lidar_gpu_reference_test tests/gpu_reference_test.cpp ${renderer_sources})
  target_compile_features(xsim_lidar_gpu_reference_test PRIVATE cxx_std_17)
  target_compile_options(xsim_lidar_gpu_reference_test PRIVATE -UNDEBUG)
  target_include_directories(xsim_lidar_gpu_reference_test PRIVATE ${renderer_includes})
  target_compile_definitions(xsim_lidar_gpu_reference_test PRIVATE ROOT_DIR="${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/marsim/")
  target_link_libraries(xsim_lidar_gpu_reference_test PRIVATE ${renderer_libraries})
endif()
install(TARGETS xsim-lidar-gpu LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/xsim)
install(FILES LICENSE source-pin.json marsim-memory-entry.patch DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu)
install(FILES ${upstream}/360camera.vs ${upstream}/camera.fs DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/marsim/include)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/" DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/source/src/xsim-lidar-gpu)
install(FILES ../xsim/sensors/lidar_gpu_module.h DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/source/src/xsim/sensors)
install(FILES ../../.xgc2/scripts/check_gpu_module_sources.py DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim-lidar-gpu/source/.xgc2/scripts)
